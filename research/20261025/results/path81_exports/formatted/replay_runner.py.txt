"""Replay stage39 formatted exports without refitting or selecting any model."""
from __future__ import annotations

import difflib
import hashlib
import json
import logging
import os
from pathlib import Path
import shutil
import subprocess
import time

R = Path('/home/xingrui/bench/quantized_graph/research_20261025')
D = Path('/home/xingrui/workspace/quantized_graph/symphony_research_20261025/research/20261025')
A = R / 'replay81_exports_formatted_v1'
PYTHON = '/home/xingrui/bench/symphony_qg/.venv/bin/python'
logging.basicConfig(level=logging.INFO, format='%(message)s')


def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def read(path):
    return json.loads(Path(path).read_text())


def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def save(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')


def main():
    A.mkdir(exist_ok=False)
    for name in ['scripts', 'logs', 'diffs']:
        (A / name).mkdir()
    files = read(R / 'results/stage39_exports_archive_filelist_v1.json')['files']
    protected = {}
    source_pairs = []
    for item in files:
        original, archived = Path(item['source']), Path(item['target'])
        need(sha(original) == item['original_sha256'], 'Successful original changed')
        protected[str(original)], protected[str(archived)] = sha(original), sha(archived)
        if archived.parent == D / 'scripts':
            diff = ''.join(difflib.unified_diff(original.read_text().splitlines(True), archived.read_text().splitlines(True), fromfile=str(original), tofile=str(archived)))
            (A / 'diffs' / (archived.name + '.diff')).write_text(diff)
    dependencies = ['path81_actual_score.cc', 'path81_actual_score.py', 'audit81_primary_metrics_v1.py',
                    'path80_score.cc', 'independent78_import.h', 'candidate81_codec.h']
    for name in dependencies:
        archived = D / 'scripts' / name
        target = A / 'scripts' / name
        protected[str(archived)] = sha(archived)
        shutil.copyfile(archived, target)
        need(sha(target) == sha(archived), 'Archive source copy differs')
        source_pairs.append({'archived': str(archived), 'archived_sha256': sha(archived),
                             'executed_copy': str(target), 'executed_copy_sha256': sha(target)})
    save(A / 'source_manifest.json', {'source_pairs': source_pairs, 'protected_input_sha256': protected,
         'path_only_edits_in_archive': ['scorer source dependencies resolved from invoked script directory, not raw root/scripts',
                                       'independent audit accepts explicit --root/--export/--output; numerical protocol unchanged']})
    env = dict(os.environ, OPENBLAS_NUM_THREADS='1', OMP_NUM_THREADS='1', PYTHONDONTWRITEBYTECODE='1')
    commands, checks = [], {}

    def run(name, argv):
        logging.info('STAGE39 START %s', name)
        start = time.time()
        log = A / 'logs' / (name + '.log')
        with log.open('x') as stream:
            process = subprocess.run(argv, cwd=R, env=env, stdout=stream, stderr=subprocess.STDOUT, check=False)
        record = {'name': name, 'argv': argv, 'cwd': str(R), 'returncode': process.returncode,
                  'elapsed_seconds': time.time() - start, 'log': str(log)}
        commands.append(record)
        save(A / 'logs' / (name + '_command_result.json'), record)
        need(process.returncode == 0, 'Replay command failed: ' + name)
        logging.info('STAGE39 PASS %s', name)

    try:
        base = [PYTHON, str(A / 'scripts/path81_actual_score.py'), '--root', str(R),
                '--candidate-manifest', str(R / 'path81_candidates_v1/manifest.json'), '--execute']
        for phase in ['training', 'validation']:
            extra = [] if phase == 'training' else ['--frozen-models', str(R / 'path81_training_v1/training_frozen_models.json'),
                                                     '--training-export-proof', str(R / 'path81_training_export_v1/proof.json')]
            run(phase, [*base, '--mode', phase, '--output', str(A / phase), *extra])
            original = R / ('path81_' + phase + '_export_v1')
            proof = read(A / phase / 'proof.json')
            old_proof = read(original / 'proof.json')
            need(read(A / phase / 'scorer_result.json') == read(original / 'scorer_result.json'), 'Scorer numeric result differs')
            hashes = {}
            for name in ['states.bin', 'queries.bin', 'scores.f32']:
                need(sha(A / phase / name) == sha(original / name), 'Numeric export bytes differ')
                hashes[name] = sha(A / phase / name)
            for key in ['protocol_sha256', 'candidate_manifest_sha256', 'training_frozen_models_sha256']:
                need(proof[key] == old_proof[key], 'Protocol/model binding differs')
            compile_argv = proof['commands'][0]['argv']
            need(str(A / phase / 'source/path81_actual_score.cc') in compile_argv, 'Wrong compilation source')
            for pair in source_pairs:
                name = Path(pair['archived']).name
                if name != 'audit81_primary_metrics_v1.py':
                    need(sha(A / phase / 'source' / name) == pair['archived_sha256'], 'Compiled source snapshot not archive')
            checks[phase] = {'all_numeric_artifacts_byte_identical': hashes,
                             'all_scorer_result_fields_equal': True, 'compiled_archived_source_verified': True,
                             'same_original_frozen_model_and_candidate_bank': True,
                             'source_proof_sha256': sha(original / 'proof.json'),
                             'formatted_proof_sha256': sha(A / phase / 'proof.json')}
        run('independent_audit', [PYTHON, str(A / 'scripts/audit81_primary_metrics_v1.py'), '--root', str(R),
                                  '--export', str(A / 'validation'), '--output', str(A / 'independent_metrics')])
        original, new = read(R / 'path81_independent_metrics_v1/report.json'), read(A / 'independent_metrics/report.json')
        allowed = {'source_sha256', 'validation_export_proof_sha256'}
        need({k: v for k, v in original.items() if k not in allowed} ==
             {k: v for k, v in new.items() if k not in allowed}, 'Independent metric fields differ')
        need(sha(R / 'path81_independent_metrics_v1/per_query.json') == sha(A / 'independent_metrics/per_query.json'), 'Perquery metric output bytes differ')
        checks['independent_audit'] = {'all_algorithm_numeric_fields_equal': True, 'per_query_bytes_equal': True,
                                       'allowed_provenance_fields': sorted(allowed)}
        for path, digest in protected.items():
            need(sha(path) == digest, 'Protected original/archive changed during replay')
        result = {'passed': True, 'stage': 39, 'sandbox': str(A), 'checks': checks, 'commands': commands,
                  'source_manifest_sha256': sha(A / 'source_manifest.json'), 'source_manifest': str(A / 'source_manifest.json'),
                  'protected_input_count': len(protected), 'protected_inputs_unchanged': True,
                  'numeric_binary_differences_allowed': False,
                  'allowed_differences': ['script and binary hashes', 'source/output/log paths', 'execution elapsed time',
                                          'proof hashes derived from source/path metadata'],
                  'archive_path_portability_edits': read(A / 'source_manifest.json')['path_only_edits_in_archive'],
                  'same_frozen_data_replay_not_new_model_selection': True,
                  'new_training': False, 'new_validation_acceptance_test': False,
                  'primary_query_retrieval': False, 'QPS_or_new_paths': False,
                  'independent_audit_not_blind_to_unsolicited_reviewer_summary': True,
                  'runner_sha256': sha(Path(__file__))}
        save(A / 'postformat_validation.json', result)
        save(R / 'results/stage39_exports_postformat_validation.json', result)
        logging.info('STAGE39 EXPORT REPLAY COMPLETE')
    except BaseException as error:
        save(A / 'failure.json', {'error': repr(error), 'commands': commands, 'checks_completed': checks})
        raise


if __name__ == '__main__':
    main()
