from __future__ import annotations

import hashlib
import json
import logging

from pathlib import Path

R = Path('/home/xingrui/bench/quantized_graph/research_20261025')
A = R / 'replay80_formatted_v1'
ARCHIVE = Path('/home/xingrui/workspace/quantized_graph/symphony_research_20261025/research/20261025')
logging.basicConfig(level=logging.INFO, format='%(message)s')


def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(4 << 20), b''):
            digest.update(block)
    return digest.hexdigest()


def read(path):
    return json.loads(Path(path).read_text())


def save(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, allow_nan=False)
        stream.write('\n')


def main():  # noqa: PLR0915 - keep the final evidence audit explicit
    failure = read(A / 'failure.json')
    need(failure['error'] == 'TypeError("string indices must be integers, not \'str\'")', 'Unexpected initial failure')
    checks = failure['checks_completed']
    need(checks['independent_actual_prefix_all1000_equal'], 'Original replay did not finish its numeric gates')
    commands = list(failure['commands'])
    need(all(item['returncode'] == 0 for item in commands), 'An executed validation command failed')
    manifest = read(A / 'source_manifest.json')
    # These are assertions against frozen outputs, not new numeric calculations.
    for name, expected in checks['scorer_all1000']['binary_artifacts_byte_equal'].items():
        need(sha(A / 'path80_native_v1' / name) == expected, 'Sandbox scorer output changed')
        need(sha(R / 'path80_native_v1' / name) == expected, 'Original scorer output changed')
    replays = ['path80_native_replay_v1', 'path80_identity_replay_v2', 'path80_synthetic_replay_v2',
               'path80_same_bin_replay_v3', 'path80_learned_replay_v2']
    for dirname in replays:
        for name in ['queries.jsonl', 'proposals.csv']:
            need(sha(A / dirname / name) == sha(R / dirname / name), 'Replay bytes changed')
    local_source = A / 'lifo/path80_lifo_control_v1.cc'
    need(sha(local_source) == sha(ARCHIVE / 'scripts/path80_lifo_control_v1.cc'), 'LIFO source is not archive copy')
    original = read(R / 'path80_lifo_control_v1/commands.json')
    need(isinstance(original, dict), 'LIFO manifest must be a mapping')
    need(len(original['commands']) == 2, 'LIFO manifest must contain compile and run')
    for path, expected in original['source_sha256'].items():
        need(sha(path) == expected, 'LIFO frozen input hash changed')
    env = {k: v for k, v in os.environ.items() if not k.startswith(('QG_', 'BENCH_', 'GTEST_'))
           and k not in ['PYTHONOPTIMIZE', 'PYTHONPATH', 'LD_PRELOAD', 'LD_LIBRARY_PATH']}
    env.update(OPENBLAS_NUM_THREADS='1', OMP_NUM_THREADS='1', PYTHONDONTWRITEBYTECODE='1')
    for name, raw in zip(['compile', 'run'], original['commands'], strict=True):
        command = [arg.replace(str(R / 'path80_lifo_control_v1'), str(A / 'lifo')) for arg in raw]
        if name == 'compile':
            need(str(local_source) in command, 'Wrong LIFO compilation source')
        log = A / 'logs' / ('lifo_' + name + '_v2.log')
        started = time.time()
        with log.open('x') as stream:
            completed = subprocess.run(command, cwd=A, env=env, stdout=stream, stderr=subprocess.STDOUT, check=False)
        record = {'name': 'lifo_' + name + '_v2', 'argv': command, 'cwd': str(A),
                  'returncode': completed.returncode, 'elapsed_seconds': time.time() - started, 'log': str(log)}
        save(A / 'logs' / ('lifo_' + name + '_v2_command_result.json'), record)
        commands.append(record)
        need(completed.returncode == 0, 'LIFO command failed: ' + name)
    lifo = read(A / 'logs/lifo_run_v2.log')
    old_lifo = read(R / 'results/path80_lifo_control_v1.json')
    need(all(old_lifo[key] == value for key, value in lifo.items()), 'LIFO numeric control differs')
    checks['production_lifo_control_equal'] = True
    for path, expected in manifest['protected_input_sha256'].items():
        need(sha(path) == expected, 'Protected original/archive/production input changed: ' + path)
    for pair in manifest['source_pairs']:
        need(sha(pair['executed_copy']) == pair['archived_sha256'], 'Executed archive copy changed')
    checks['final_rehash_of_all_protected_sources_and_numeric_binaries'] = True
    result = {'passed': True, 'stage': 36, 'sandbox': str(A), 'checks': checks, 'commands': commands,
              'source_manifest': str(A / 'source_manifest.json'), 'source_manifest_sha256': sha(A / 'source_manifest.json'),
              'protected_input_count': len(manifest['protected_input_sha256']), 'protected_inputs_unchanged': True,
              'production_headers_unchanged': True, 'bench_success_sources_and_evidence_unchanged': True,
              'allowed_differences': ['script/binary SHA due formatting and source paths', 'output/source/log paths',
                                      'wall-clock durations (validation only; no QPS)',
                                      'proof hashes caused only by provenance changes',
                                      'CLI print replaced with logging; assertions split without changing checks',
                                      'explicit strict=False preserves comparator shorter-sequence zip behavior'],
              'archive_semantic_edits': ['format/import ordering', 'CLI logging', 'split conjunction asserts', 'dict literals',
                                         'unused binding removal while preserving call', 'local complexity noqa avoids algorithm refactoring',
                                         'explicit non-strict zip'],
              'no_performance_timing': True, 'no_retrieval_or_training': True,
              'trace80_primary_archived_but_not_rerun': True, 'no_unfinished_mechanism_A_B_inputs': True,
              'first_orchestrator_failure': {'path': str(A / 'failure.json'), 'sha256': sha(A / 'failure.json'),
                  'cause': 'LIFO command manifest is a dict containing commands, not a list of command records',
                  'all_numeric_replays_had_already_passed': True,
                  'recovery': 'Separate v2 continuation reads the dict, runs remaining LIFO control and rehashes evidence; no original overwritten'},
              'runners': {str(R / 'scripts/stage36_postformat_replay.py'): sha(R / 'scripts/stage36_postformat_replay.py'),
                          str(Path(__file__)): sha(Path(__file__))}}
    save(A / 'stage36_postformat_validation.json', result)
    save(R / 'results/stage36_postformat_validation.json', result)
    logging.info('PASS: full1000 stage36 post-format validation with preserved first orchestrator failure')


if __name__ == '__main__':
    main()
